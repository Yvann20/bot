def log(text):
    now = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
    with open(LOG_FILE, "a") as f:
        f.write(f"[{now}] {text}\n")
    print(f"[{now}] {text}")

def save_channels(origin, dest):
    with open(CHANNELS_FILE, "w") as f:
        f.write(f"{origin}\n{dest}\n")

def load_channels():
    if os.path.exists(CHANNELS_FILE):
        with open(CHANNELS_FILE) as f:
            lines = f.read().splitlines()
            if len(lines) >= 2:
                return lines[0], lines[1]
    return "", ""

def set_source_channel(ch):

def set_dest_channel(ch):

def sanitize_channel(text):
    text = text.strip()
    m = re.match(r'(https?://)?t\.me/([a-zA-Z0-9_]+)', text)
    if m:
        return '@' + m.group(2)
    if text.startswith('@'):
        return text
    return '@' + text.lstrip('@/')

def init_db():
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute('''CREATE TABLE IF NOT EXISTS enviados (message_id INTEGER PRIMARY KEY)''')
    cursor.execute('''CREATE TABLE IF NOT EXISTS audit_log (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        admin TEXT, ip TEXT, acao TEXT, status TEXT,
        datahora DATETIME DEFAULT CURRENT_TIMESTAMP)''')
    cursor.execute('''CREATE TABLE IF NOT EXISTS blacklist (
        palavra TEXT PRIMARY KEY)''')
    cursor.execute('''CREATE TABLE IF NOT EXISTS erros (
        id INTEGER PRIMARY KEY AUTOINCREMENT, 
        datahora DATETIME DEFAULT CURRENT_TIMESTAMP,
        error_text TEXT, traceback TEXT
    )''')
    conn.commit()
    conn.close()
    load_blacklist()

def log_auditoria(admin, ip, acao, status):
    try:
        conn = sqlite3.connect(DB_NAME)
        cursor = conn.cursor()
        cursor.execute("INSERT INTO audit_log (admin, ip, acao, status) VALUES (?, ?, ?, ?)",
                       (admin, ip, acao, status))
        conn.commit()
        conn.close()
    except Exception as e:
        log(f"Erro ao registrar auditoria: {e}")

def blacklisted(texto):
    texto = (texto or "").lower()
    for palavra in BLACKLIST:
        if palavra in texto:
            return True
    return False

def add_blacklist(palavra):
    palavra = palavra.lower().strip()
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("INSERT OR IGNORE INTO blacklist (palavra) VALUES (?)", (palavra,))
    conn.commit()
    conn.close()
    BLACKLIST.add(palavra)

def remove_blacklist(palavra):
    palavra = palavra.lower().strip()
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("DELETE FROM blacklist WHERE palavra = ?", (palavra,))
    conn.commit()
    conn.close()
    BLACKLIST.discard(palavra)

def load_blacklist():
    global BLACKLIST
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("SELECT palavra FROM blacklist")
    BLACKLIST = set([row[0] for row in cursor.fetchall()])
    conn.close()

def save_erro(error_text, tb):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("INSERT INTO erros (error_text, traceback) VALUES (?, ?)", (error_text, tb))
    conn.commit()
    conn.close()

def is_real_video(msg):
    # Se o vídeo tem texto e não cai na blacklist, é real.
    return (
        msg.video and
        msg.media and
        hasattr(msg.media, 'document') and
        not getattr(msg, 'sticker', False) and
        not getattr(msg, 'gif', False) and
        not blacklisted(getattr(msg, 'text', ''))
    )

def is_source_channel(event):
    ch_username = getattr(event.chat, 'username', None)
    if ch_username and ("@" + ch_username == source_channel):
        return True
    if not ch_username and str(event.chat_id) == source_channel.lstrip("@"):
        return True
    return False

def backup_db():
    today = datetime.now().strftime('%Y-%m-%d_%H-%M-%S')
    backup_name = os.path.join(BACKUP_DIR, f"encaminhados_{today}.db")
    shutil.copy2(DB_NAME, backup_name)
    backups = sorted([f for f in os.listdir(BACKUP_DIR) if f.endswith(".db")])
    if len(backups) > 10:
        os.remove(os.path.join(BACKUP_DIR, backups[0]))

def reset_contadores():
    global contagem_hora, contagem_dia, hora_atual, dia_atual
    now = datetime.now()
    if now.hour != hora_atual:
        contagem_hora = 0
        hora_atual = now.hour
    if now.day != dia_atual:
        contagem_dia = 0
        dia_atual = now.day

@app.route('/thumb/<msgid>')
def thumb(msgid):
    async def load_thumb():
        msg = await user_client.get_messages(source_channel, ids=int(msgid))
        if hasattr(msg, 'photo') or (hasattr(msg, "video") and msg.video):
            img_path = f"cache/thumb_{msgid}.jpg"
            await msg.download_media(file=img_path, thumb=-1)
            return img_path
        return None
    fut = asyncio.run_coroutine_threadsafe(load_thumb(), loop)
    path = fut.result(timeout=20)
    if path and os.path.exists(path):
        return send_file(path, mimetype="image/jpeg")
    return "Sem thumb", 404

async def encaminhar_videos_web():
    global enviados, erros, enviando, contagem_hora, contagem_dia
    if not source_channel or not dest_channel:
        log("⚠️ Defina canal de origem e destino antes!")
        return
    try:
        all_messages = await user_client.get_messages(source_channel, limit=None)
    except Exception as e:
        log(f"❌ Erro ao acessar o canal de origem: {e}")
        save_erro(str(e), traceback.format_exc())
        notificar_webhook(f"Erro ao acessar canal origem: {e}")
        enviando = False
        return
    videos = [msg for msg in all_messages if is_real_video(msg)]
    if not videos:
        log("Nenhum vídeo encontrado no canal de origem.")
        return
    for i in range(0, len(videos), 10):
        if not enviando:
            log("Encaminhamento parado pelo usuário.")
            break
        reset_contadores()
        if contagem_hora >= limite_hora or contagem_dia >= limite_dia:
            log("⏳ Limite de encaminhamento atingido. Aguardando 1h...")
            await asyncio.sleep(3600)
            reset_contadores()
        lote = videos[i:i+10]
        for msg in lote:
            if ja_foi_enviado(msg.id):
                continue
            if simulacao:
                log(f"[SIMULAÇÃO] Encaminharia: {msg.id} - {getattr(msg, 'text', '')[:40]}")
                continue
            try:
                await user_client.forward_messages(dest_channel, msg, silent=True)  # Envio stealth
                marcar_como_enviado(msg.id)
                enviados += 1
                contagem_hora += 1
                contagem_dia += 1
                log(f"✅ Forwarded: {msg.id}")
                await asyncio.sleep(random.randint(2, 4))
            except Exception as e:
                erros += 1
                log(f"❌ Erro no {msg.id}: {e}")
                save_erro(str(e), traceback.format_exc())
                notificar_webhook(f"Erro grave no encaminhamento: {e}")
                socketio.emit("erro", {"msg": str(e)}, broadcast=True)
                if "Too Many Requests" in str(e) or "Flood" in str(e):
                    log("⏳ Flood control ativado, aguardando 60s...")
                    await asyncio.sleep(60)
        espera = random.randint(20, 30)
        log(f"Lote {i//10+1} encaminhado. Aguardando {espera}s...")
        await asyncio.sleep(espera)
    log(f"✅ Finalizado. Total encaminhados: {enviados} | Erros: {erros}")

def ja_foi_enviado(message_id):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("SELECT 1 FROM enviados WHERE message_id = ?", (message_id,))
    result = cursor.fetchone()
    conn.close()
    return result is not None

def marcar_como_enviado(message_id):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("INSERT OR IGNORE INTO enviados (message_id) VALUES (?)", (message_id,))
    conn.commit()
    conn.close()

async def exportar_ids(event=None):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("SELECT message_id FROM enviados")
    ids = cursor.fetchall()
    conn.close()
    filename = "ids_enviados.txt"
    with open(filename, "w") as f:
        for id_tuple in ids:
            f.write(f"{id_tuple[0]}\n")
    if event:
        await event.respond("Exportação concluída:", file=filename)
    else:
        log("Exportação concluída.")

async def buscar_video_nome(event, termo):
    if not source_channel:
        await event.respond("⚠️ Defina o canal de origem antes!")
        return
    await event.respond(f"🔍 Buscando vídeo(s) com: `{termo}` ...")
    messages = await user_client.get_messages(source_channel, limit=500)
    results = [msg for msg in messages if is_real_video(msg) and msg.text and termo.lower() in msg.text.lower()]
    if not results:
        await event.respond("Nenhum vídeo encontrado com esse termo.")
    else:
        txt = "\n".join(f"- ID: {msg.id} | Texto: {msg.text[:30]}" for msg in results[:10])
        await event.respond(f"Encontrei {len(results)} vídeo(s):\n{txt}\n\nDeseja encaminhar algum? Envie o ID.")

async def encaminhar_por_id(event, id_str):
    if not source_channel or not dest_channel:
        await event.respond("Defina o canal de origem e destino antes!")
        return
    try:
        msg = await user_client.get_messages(source_channel, ids=int(id_str))
        if not is_real_video(msg):
            await event.respond("Esse ID não corresponde a um vídeo válido.")
            return
        if simulacao:
            await event.respond(f"[SIMULAÇÃO] Encaminharia o vídeo ID {msg.id}")
            return
        await user_client.forward_messages(dest_channel, msg, silent=True)
        marcar_como_enviado(msg.id)
        await event.respond("Vídeo encaminhado com sucesso!")
    except Exception as e:
        await event.respond(f"Erro ao encaminhar: {e}")
        save_erro(str(e), traceback.format_exc())
        notificar_webhook(f"Erro ao encaminhar por ID: {e}")

@bot.on(events.CallbackQuery)
async def callbacks(event):
    global enviando, enviados, erros, auto_mode, auto_paused, waiting_for_channel, waiting_for_search, simulacao
    if event.sender_id not in admin_user_ids:
        await event.answer("Acesso não autorizado.", alert=True)
        return
    data = event.data.decode('utf-8')

    if data == "start_forward":
        if enviando:
            await event.answer("Já está enviando.", alert=True)
            return
        if not source_channel or not dest_channel:
            await event.answer("Defina origem e destino primeiro!", alert=True)
            return
        simulacao = False
        enviando = True
        enviados = 0
        erros = 0
        await event.answer("Iniciando manual...", alert=True)
        asyncio.create_task(encaminhar_videos_web())
    elif data == "stop_forward":
        enviando = False
        await event.answer("Encaminhamento manual será parado.", alert=True)
    elif data == "auto_on":
        if not source_channel or not dest_channel:
            await event.answer("Defina origem e destino primeiro!", alert=True)
            return
        auto_mode = True
        auto_paused = False
        await event.answer("Auto-encaminhamento ativado!", alert=True)
        await event.respond("🔄 O bot agora encaminhará automaticamente todo novo vídeo do canal de origem.")
    elif data == "auto_off":
        auto_mode = False
        await event.answer("Auto-encaminhamento desativado!", alert=True)
        await event.respond("⏹️ Auto-encaminhamento parado.")
    elif data == "pause_auto":
        if not auto_mode:
            await event.answer("Auto-encaminhar já está desativado.", alert=True)
        else:
            auto_paused = True
            await event.answer("Auto-encaminhamento PAUSADO.", alert=True)
    elif data == "resume_auto":
        if not auto_mode:
            await event.answer("Ative o auto-encaminhar antes.", alert=True)
        else:
            auto_paused = False
            await event.answer("Auto-encaminhamento RETOMADO.", alert=True)
    elif data == "sim_on":
        simulacao = True
        await event.answer("Modo simulação ATIVADO!", alert=True)
    elif data == "sim_off":
        simulacao = False
        await event.answer("Modo simulação DESATIVADO!", alert=True)
    elif data == "search_name":
        waiting_for_search = True
        await event.respond("Envie a palavra ou trecho para buscar no nome/texto do vídeo:")
    elif data == "clear_ids":
        limpar_ids()
        await event.answer("IDs limpos!", alert=True)
        await event.respond("🗑️ IDs do banco foram limpos, pode encaminhar todos de novo.")
    elif data == "relatorio":
        await enviar_relatorio(event)
        await event.answer("Relatório enviado!", alert=True)
    elif data == "set_origem":
        waiting_for_channel = 'origem'
        await event.respond("Envie agora o @usuario ou link do CANAL **DE ORIGEM**:")
    elif data == "set_destino":
        waiting_for_channel = 'destino'
        await event.respond("Envie agora o @usuario ou link do CANAL **DE DESTINO**:")
    elif data == "status":
        await start_menu(event)
        await event.answer("Status atualizado!", alert=True)
    elif data == "export":
        await exportar_ids(event)
        await event.answer("IDs exportados!", alert=True)

@bot.on(events.NewMessage)
async def on_text_commands(event):
    global waiting_for_channel, waiting_for_search
    if event.sender_id not in admin_user_ids:
        return
    if waiting_for_channel:
        channel_username = sanitize_channel(event.text)
        if waiting_for_channel == "origem":
            set_source_channel(channel_username)
            await event.respond(f"Canal de ORIGEM definido: {source_channel}")
        elif waiting_for_channel == "destino":
            set_dest_channel(channel_username)
            await event.respond(f"Canal de DESTINO definido: {dest_channel}")
        waiting_for_channel = None
        return
    if waiting_for_search:
        termo = event.text.strip()
        waiting_for_search = False
        await buscar_video_nome(event, termo)
        return
    if event.text.strip().isdigit() and source_channel and dest_channel:
        await encaminhar_por_id(event, event.text.strip())

async def auto_forward_new_video(event):
    global auto_mode, source_channel, dest_channel, auto_paused, contagem_hora, contagem_dia
    if auto_mode and not auto_paused and source_channel and dest_channel:
        try:
            if is_source_channel(event) and is_real_video(event.message):
                reset_contadores()

                    await user_client.forward_messages(dest_channel, msg, silent=True)
